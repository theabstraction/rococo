Containers

Sexy has three built-in containers: array, list and map. They are declared like variables, typed by element, and shared by reference when assigned or passed.

Arrays

(array Int32 a 4) creates an empty array with room for four elements. Push appends, (a i) reads, Set writes and Length counts. Indices start at 0. foreach x # a visits the elements in order.

Sexy
(function Main -> (Int32 result):
    (array Int32 a 4)                  // capacity 4, length 0
    (a.Push 10)
    (a.Push 20)
    (a.Push 30)
    (Int32 first = (a 0))              // indices start at 0
    (a.Set 1 25)
    (Int32 sum = 0)
    (foreach x # a
        (sum += x)
    )
    (result = sum + first)
)
Lua 5.4
local a = {}
a[#a + 1] = 10
a[#a + 1] = 20
a[#a + 1] = 30
local first = a[1]        -- Lua sequences start at 1
a[2] = 25
local sum = 0
for _, x in ipairs(a) do sum = sum + x end
assert(sum + first == 75)
assert(a[4] == nil)       -- reading past the end is not an error in Lua

Result Main returns 75

Reading past the length throws, where Lua gives nil and Python raises IndexError:

Sexy
(function Main -> (Int32 result):
    (array Int32 a 4)
    (a.Push 10)
    (result = (a 3))                   // length is 1: throws
)
Python 3
a = [10]
try:
    a[3]
    raise SystemExit("unreachable")
except IndexError:
    pass

Error Array.Get failed: the index was out of range

Arrays of structs

Pushing a struct names the type and gives the member values in parentheses. Reading an element into a variable copies it.

Sexy
(struct Vec2i (Int32 x y))

(function Main -> (Int32 result):
    (array Vec2i points 8)
    (points.Push Vec2i (1 2))
    (points.Push Vec2i (30 4))
    (foreach p # points
        (result += p.x)
    )
    (Vec2i last = (points 1))
    (result += (last.y + 2))
)
C++20
#include <cassert>
#include <vector>

struct Vec2i { int x, y; };

int main()
{
    std::vector<Vec2i> points;
    points.reserve(8);
    points.push_back({ 1, 2 });
    points.push_back({ 30, 4 });
    int result = 0;
    for (const Vec2i& p : points) result += p.x;
    Vec2i last = points[1];         // copy, as in Sexy
    result += last.y + 2;
    assert(result == 37);
}

Result Main returns 37

Lists

A list is doubly linked. Append and Prepend add at either end. foreach over a list binds a node, not the value: read the element with n.Value.

Sexy
(function Main -> (Int32 result):
    (list Int32 l)
    (l.Append 2)
    (l.Append 3)
    (l.Prepend 1)
    (l.Append 4)
    (foreach n # l
        (Int32 v = n.Value)
        (result = (result * 10) + v)
    )
)
C++20
#include <cassert>
#include <list>

int main()
{
    std::list<int> l;
    l.push_back(2);
    l.push_back(3);
    l.push_front(1);
    l.push_back(4);
    int result = 0;
    for (int v : l) result = result * 10 + v;
    assert(result == 1234);
}

Result Main returns 1234

Removing elements during foreach

A node can remove itself with n.Pop while the list is being enumerated, and the loop carries on with the next element. Every element is visited once, so there is no erase-and-advance pattern to get wrong.

Sexy
(function Main -> (Int32 result):
    (list Int32 l)
    (l.Append 1)
    (l.Append 2)
    (l.Append 3)
    (l.Append 4)
    (Int32 visited = 0)
    (foreach n # l
        (visited += 1)
        (Int32 v = n.Value)
        (if (v == 2) (n.Pop))
        (if (v == 3) (n.Pop))
    )
    (Int32 len = l.Length)
    (result = (visited * 1000) + len)
)
C++20
#include <cassert>
#include <list>

int main()
{
    std::list<int> l = { 1, 2, 3, 4 };
    int visited = 0;
    for (auto it = l.begin(); it != l.end(); )   // a range-for would be invalidated by erase
    {
        ++visited;
        if (*it == 2 || *it == 3) it = l.erase(it);
        else ++it;
    }
    assert(visited * 1000 + static_cast<int>(l.size()) == 4002);
}

Result Main returns 4002

Maps

(map IString Int32 ages) maps keys to values. Keys can be IString, Int32, Float32 or Float64. Insert adds or overwrites. (ages "Ada") returns a node; n.Exists says whether the key was found.

Sexy
(using Sys.Type)

(function Main -> (Int32 result):
    (map IString Int32 ages)
    (ages.Insert "Ada" 36)
    (ages.Insert "Alan" 41)
    (ages.Insert "Ada" 37)             // Insert overwrites
    (node n = (ages "Ada"))
    (Int32 adaAge = n.Value)
    (node missing = (ages "Grace"))
    (if (not missing.Exists) (result = 1000))
    (result += (adaAge - 30))
)
Python 3
ages: dict[str, int] = {}
ages["Ada"] = 36
ages["Alan"] = 41
ages["Ada"] = 37                       # assignment overwrites
result = 0
if "Grace" not in ages:
    result = 1000
result += ages["Ada"] - 30
assert result == 1007

Result Main returns 1007

Reading n.Value for a missing key throws. C++'s operator[] would quietly insert a zero instead:

Sexy
(using Sys.Type)

(function Main -> (Int32 result):
    (map IString Int32 ages)
    (ages.Insert "Ada" 36)
    (node n = (ages "Grace"))
    (result = n.Value)                 // no such key: throws
)
C++20
#include <cassert>
#include <map>
#include <stdexcept>
#include <string>

int main()
{
    std::map<std::string, int> ages = { { "Ada", 36 } };
    int silent = ages["Grace"];            // operator[] inserts a zero entry instead of failing
    assert(silent == 0 && ages.size() == 2);
    try
    {
        (void) ages.at("Alan");            // at() throws, like Sexy's n.Value
        return 1;
    }
    catch (const std::out_of_range&) {}
}

Error MapNodeGet32 failed. The node did not represent an entry in the map

Iterating over a map

foreach over a map binds a node with Key and Value. The examples only rely on visiting each entry once, not on the order.

Sexy
(function Main -> (Int32 result):
    (map Int32 Int32 squares)
    (squares.Insert 2 4)
    (squares.Insert 3 9)
    (squares.Insert 8 64)
    (foreach n # squares
        (Int32 k = n.Key)
        (Int32 v = n.Value)
        (result += v)
        (result -= k)
    )
    (Int32 len = squares.Length)
    (result += (len + 10))
)
Python 3
squares = {2: 4, 3: 9, 8: 64}
result = 0
for k, v in squares.items():
    result += v
    result -= k
result += len(squares) + 10
assert result == 77

Result Main returns 77

As with lists, an entry can be popped during the loop. Python refuses to let a dict change size while it is being iterated:

Sexy
(function Main -> (Int32 result):
    (map Int32 Int32 m)
    (m.Insert 1 10)
    (m.Insert 2 20)
    (m.Insert 3 30)
    (foreach n # m
        (Int32 k = n.Key)
        (if (k == 2) (n.Pop))
    )
    (result = m.Length)
)
Python 3
m = {1: 10, 2: 20, 3: 30}
try:
    for k in m:
        if k == 2:
            del m[k]                     # RuntimeError: dictionary changed size during iteration
    raise SystemExit("unreachable")
except RuntimeError:
    pass
m = {1: 10, 2: 20, 3: 30}
for k in list(m):                        # the usual workaround: iterate over a copy of the keys
    if k == 2:
        del m[k]
assert len(m) == 2

Result Main returns 2
